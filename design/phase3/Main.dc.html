-- Customers created since the last release, with their order counts
DECLARE @since datetime2 = '2025-01-01';
 
SELECT TOP (100)
c.CustomerId,
c.Name,
c.Email,
c.Phone,
c.CreatedDate,
COUNT(o.OrderId) AS Orders
FROM dbo.Customer AS c
LEFT JOIN dbo.[Order] AS o ON o.CustomerId = c.CustomerId
WHERE c.CreatedDate >= @since
GROUP BY c.CustomerId, c.Name, c.Email, c.Phone, c.CreatedDate
ORDER BY c.CreatedDate DESC;
